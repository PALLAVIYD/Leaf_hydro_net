import tensorflow as tf
import tensorflow_datasets as tfds
import matplotlib.pyplot as plt
import seaborn as sns
import numpy as np
from sklearn.metrics import confusion_matrix, classification_report

# ==========================================================
# 3️⃣ Load PlantVillage Dataset (Auto Download)
# ==========================================================
(ds_train, ds_test), ds_info = tfds.load(
    'plant_village',
    split=['train[:80%]', 'train[80%:]'],
    as_supervised=True,
    with_info=True
)

NUM_CLASSES = ds_info.features['label'].num_classes
IMG_SIZE = 128
BATCH_SIZE = 32

print("Number of Classes:", NUM_CLASSES)

# ==========================================================
# 4️⃣ Preprocess Dataset
# ==========================================================
def preprocess(image, label):
    image = tf.image.resize(image, (IMG_SIZE, IMG_SIZE))
    image = image / 255.0
    return image, label

ds_train = ds_train.map(preprocess).shuffle(1000).batch(BATCH_SIZE).prefetch(tf.data.AUTOTUNE)
ds_test = ds_test.map(preprocess).batch(BATCH_SIZE).prefetch(tf.data.AUTOTUNE)

# ==========================================================
# 5️⃣ Build CNN Model
# ==========================================================
model = tf.keras.Sequential([
    tf.keras.layers.Conv2D(32, (3,3), activation='relu', input_shape=(IMG_SIZE, IMG_SIZE, 3)),
    tf.keras.layers.MaxPooling2D(),
    tf.keras.layers.Conv2D(64, (3,3), activation='relu'),
    tf.keras.layers.MaxPooling2D(),
    tf.keras.layers.Conv2D(128, (3,3), activation='relu'),
    tf.keras.layers.MaxPooling2D(),
    tf.keras.layers.Flatten(),
    tf.keras.layers.Dense(256, activation='relu'),
    tf.keras.layers.Dense(NUM_CLASSES, activation='softmax')
])

model.compile(
    optimizer='adam',
    loss='sparse_categorical_crossentropy',
    metrics=['accuracy']
)

model.summary()

# ==========================================================
# 6️⃣ Train Model
# ==========================================================
history = model.fit(ds_train, validation_data=ds_test, epochs=5)

# ==========================================================
# 7️⃣ Evaluate Model
# ==========================================================
test_loss, test_acc = model.evaluate(ds_test)
print("\nTest Accuracy:", round(test_acc*100,2), "%")

# ==========================================================
# 8️⃣ Show Sample Predictions
# ==========================================================
class_names = ds_info.features['label'].names

for images, labels in ds_test.take(1):
    predictions = model.predict(images)
    preds = np.argmax(predictions, axis=1)

plt.figure(figsize=(12,6))
for i in range(6):
    plt.subplot(2,3,i+1)
    plt.imshow(images[i])
    plt.title(f"True: {class_names[int(labels[i])]}\nPred: {class_names[int(preds[i])]}")
    plt.axis("off")
plt.show()

# ==========================================================
# 9️⃣ Confusion Matrix
# ==========================================================
y_true = []
y_pred = []

for images, labels in ds_test:
    preds = model.predict(images)
    preds = np.argmax(preds, axis=1)
    y_true.extend(labels.numpy())
    y_pred.extend(preds)

cm = confusion_matrix(y_true, y_pred)

plt.figure(figsize=(8,6))

plt.title("Confusion Matrix")
plt.xlabel("Predicted")

plt.show()

def extract():
    return spark_df

def load_raw(df):
    df.createOrReplaceTempView("raw_table")
    return df

def validate(df):
    if df.count() == 0:
        raise Exception("Validation failed")
    return df

def transform():
    return spark.sql("""
    SELECT *, revenue*1.0 AS revenue_usd FROM raw_table
    """)

def aggregate(df):
    return df.groupBy("country").sum("revenue_usd")

def load_warehouse(df):
    df.show()
    return df

[CELL type=code]
class GNN(torch.nn.Module):
    def __init__(self):
        super().__init__()
        self.conv1 = GCNConv(4,16)
        self.conv2 = GCNConv(16,2)

    def forward(self,x,edge_index):
        x = self.conv1(x,edge_index)
        x = F.relu(x)
        x = self.conv2(x,edge_index)
        return x

gnn = GNN()

for _ in range(200):
    opt.zero_grad()
    out = gnn(data_gnn.x,data_gnn.edge_index)
    loss = F.cross_entropy(out,labels)
    loss.backward()
    opt.step()

class PipelineEnv(gym.Env):

    def __init__(self):
        super().__init__()
        self.action_space = spaces.Discrete(4)
        self.observation_space = spaces.Box(0,1,(5,),dtype=np.float32)

    def reset(self,seed=None,options=None):
        return np.random.rand(5).astype(np.float32),{}

    def step(self,action):
        reward = random.random()
        return np.random.rand(5),reward,False,False,{}

env = PipelineEnv()
rl = PPO("MlpPolicy",env,verbose=0)
rl.learn(5000)

def show_dag(status):

    clear_output(wait=True)

    colors = []
    for node in pipeline_dag.nodes():
        if status[node]=="pending":
            colors.append("blue")
        elif status[node]=="running":
            colors.append("yellow")
        elif status[node]=="success":
            colors.append("green")
        else:
            colors.append("red")

    plt.figure(figsize=(6,4))
    nx.draw(pipeline_dag,with_labels=True,node_color=colors,node_size=3000)
    plt.show()
    time.sleep(1)

[CELL type=code]
def run_pipeline():

    status = {n:"pending" for n in pipeline_dag.nodes()}
    data = None

    order = list(nx.topological_sort(pipeline_dag))

    for task in order:

        status[task] = "running"
        show_dag(status)

        try:

            sample = [[random.randint(0,5),10,0.5,0.5]]

            if model.predict(sample)[0] == 1:

                print("⚠️ Failure predicted")

                out = gnn(data_gnn.x,data_gnn.edge_index)
                root = list(mapping.keys())[out[:,1].argmax().item()]

                print("Root cause:",root)

                action = int(rl.predict(np.random.rand(5))[0])

                print("Healing:",["Retry","Reroute","Scale","Skip"][action])

            func = task_map[task]

            if task == "extract":
                data = func()
            elif task == "transform":
                data = func()
            else:
                data = func(data)

            status[task] = "success"

        except:
            status[task] = "failed"
            break

        show_dag(status)

    return data
